echo "SkiffOS on Allwinner R8 (C.H.I.P. NAND boot)"

setenv dtb_name "sun5i-r8-chip.dtb"
setenv condev "console=ttyS0,115200n8 console=tty1"

# setenv silent 1
setenv kernel_addr_r "0x42000000"
setenv fdt_addr_r    "0x43000000"

# Boot from NAND UBIFS (rootfs volume).
setenv skiff_nand_boot 'nand slc-mode on; mtdparts; ubi part UBI; ubifsmount ubi0:rootfs; ubifsload ${fdt_addr_r} /boot/${dtb_name}; ubifsload ${kernel_addr_r} /boot/zImage; setenv bootargs ubi.mtd=2 root=ubi0:rootfs rootfstype=ubifs rw init=/boot/skiff-init/skiff-init-squashfs ${condev} net.ifnames=0; bootz ${kernel_addr_r} - ${fdt_addr_r}'

run skiff_nand_boot
