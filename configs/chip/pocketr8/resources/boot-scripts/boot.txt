echo "SkiffOS on Allwinner R8 (Pocket C.H.I.P.)"

setenv dtb_name "sun5i-r8-pocketchip.dtb"
setenv condev "console=ttyS0,115200n8 console=tty1"

# setenv silent 1
setenv verify 0
setenv bootlogo "false"

setenv kernel_addr_r "0x42000000"
setenv fdt_addr_r   "0x43000000"

# use /init script to deferred mount /
setenv bootmem "root=/dev/mmcblk0p1 rootwait rw init=/boot/skiff-init/skiff-init-squashfs"

# fbcon=map:0 assigns the framebuffer console to the first display (Pocket CHIP LCD)
setenv baseconfig "fsck.repair=yes net.ifnames=0 fbcon=map:0"

# final boot args
setenv bootargs "${bootmem} ${condev} ${baseconfig}"

echo "Loading zImage..."
ext4load mmc 0:1 ${kernel_addr_r} /boot/zImage

echo "Loading ${dtb_name}..."
ext4load mmc 0:1 ${fdt_addr_r} /boot/${dtb_name}

echo "Booting with bootargs: ${bootargs}"
bootz ${kernel_addr_r} - ${fdt_addr_r}
